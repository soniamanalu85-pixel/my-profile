  <script>
    function tampilkanFoto(event) {
      const file = event.target.files[0];
      const img = document.getElementById("tFoto");
      if (file) {
        img.src = URL.createObjectURL(file);
        img.style.display = "block";
      }
    }

    function perbaruiCV() {
      document.getElementById("tNama").innerText = document.getElementById("fNama").value;
      document.getElementById("tKontak").innerText = document.getElementById("fKontak").value;
      document.getElementById("tRingkasan").innerText = document.getElementById("fRingkasan").value;
      document.getElementById("tPendidikan").innerText = document.getElementById("fPendidikan").value;
      document.getElementById("tPengalaman").innerText = document.getElementById("fPengalaman").value;
      document.getElementById("tKeahlian").innerText = document.getElementById("fKeahlian").value;
    }

    function cetakPDF() {
      perbaruiCV();
      window.print();
    }
